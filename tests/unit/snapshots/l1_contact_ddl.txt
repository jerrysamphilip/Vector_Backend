CREATE TABLE prospects (
	prospect_id VARCHAR(36) NOT NULL, 
	tenant_id VARCHAR(36) NOT NULL, 
	first_name VARCHAR(100), 
	last_name VARCHAR(100), 
	email VARCHAR(255) NOT NULL, 
	email_type VARCHAR(50), 
	email_provider VARCHAR(100), 
	phone VARCHAR(50), 
	mobile_phone VARCHAR(50), 
	owner_id VARCHAR(36), 
	account_id VARCHAR(36), 
	lifecycle_stage VARCHAR(30), 
	lead_status VARCHAR(30), 
	lead_source VARCHAR(100), 
	legal_basis VARCHAR(40), 
	deleted_at DATETIME(6), 
	deleted_by VARCHAR(36), 
	merged_into_id VARCHAR(36), 
	search_text VARCHAR(1100) GENERATED ALWAYS AS (LOWER(CONCAT_WS(' ', first_name, last_name, email, company_name, designation, REGEXP_REPLACE(COALESCE(phone, ''), '[^0-9]', ''), REGEXP_REPLACE(COALESCE(mobile_phone, ''), '[^0-9]', '')))) STORED, 
	tags JSON, 
	custom_fields JSON, 
	company_name VARCHAR(255), 
	designation VARCHAR(150), 
	industry VARCHAR(100), 
	emp_band VARCHAR(50), 
	linkedin_url VARCHAR(500), 
	linkedin_data JSON, 
	poc_city VARCHAR(100), 
	poc_state VARCHAR(100), 
	poc_country VARCHAR(100), 
	timezone VARCHAR(50), 
	consent_status VARCHAR(50), 
	consent_source VARCHAR(100), 
	consent_timestamp DATETIME(6), 
	score SMALLINT, 
	fit_score SMALLINT, 
	engagement_score SMALLINT, 
	score_grade VARCHAR(1), 
	scored_at DATETIME(6), 
	is_valid_email BOOL, 
	is_anonymized BOOL, 
	anonymized_at DATETIME(6), 
	created_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6), 
	updated_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6), 
	row_version INTEGER NOT NULL DEFAULT '1', 
	PRIMARY KEY (prospect_id), 
	CONSTRAINT uq_tenant_email UNIQUE (tenant_id, email), 
	CONSTRAINT uq_prospects_tenant_prospect UNIQUE (tenant_id, prospect_id), 
	FOREIGN KEY(tenant_id) REFERENCES tenants (tenant_id), 
	FOREIGN KEY(owner_id) REFERENCES users (user_id), 
	FOREIGN KEY(account_id) REFERENCES accounts (account_id)
)ENGINE=InnoDB CHARSET=utf8mb4 COLLATE utf8mb4_0900_ai_ci
