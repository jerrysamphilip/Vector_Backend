CREATE TABLE users (
	user_id VARCHAR(36) NOT NULL, 
	tenant_id VARCHAR(36), 
	first_name VARCHAR(100) NOT NULL, 
	last_name VARCHAR(100) NOT NULL, 
	email VARCHAR(255) NOT NULL, 
	`role` VARCHAR(50) NOT NULL, 
	status VARCHAR(50), 
	active_email VARCHAR(255) GENERATED ALWAYS AS (CASE WHEN status IN ('ACTIVE','INVITED') THEN email ELSE NULL END) STORED, 
	password_hash VARCHAR(255), 
	auth_provider VARCHAR(50), 
	google_id VARCHAR(255), 
	email_verified BOOL, 
	avatar_url VARCHAR(500), 
	invited_by VARCHAR(36), 
	custom_permissions JSON, 
	last_login_at DATETIME(6), 
	created_at DATETIME(6) DEFAULT CURRENT_TIMESTAMP(6), 
	sales_level INTEGER, 
	manager_id VARCHAR(36), 
	notify_email BOOL NOT NULL DEFAULT '1', 
	mfa_enabled BOOL NOT NULL DEFAULT false, 
	mfa_secret VARCHAR(1024), 
	mfa_recovery_codes TEXT, 
	mfa_enabled_at DATETIME(6), 
	mfa_last_step BIGINT, 
	PRIMARY KEY (user_id), 
	CONSTRAINT ck_users_status CHECK (status IN ('ACTIVE', 'INVITED', 'INACTIVE', 'SUSPENDED')), 
	CONSTRAINT uq_user_tenant_email UNIQUE (tenant_id, email), 
	FOREIGN KEY(tenant_id) REFERENCES tenants (tenant_id), 
	UNIQUE (google_id), 
	FOREIGN KEY(invited_by) REFERENCES users (user_id), 
	FOREIGN KEY(manager_id) REFERENCES users (user_id)
)ENGINE=InnoDB CHARSET=utf8mb4 COLLATE utf8mb4_0900_ai_ci
